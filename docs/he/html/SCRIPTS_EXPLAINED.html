מסמך הסבר לצוות. נכתב 2026-10-24 מול הקוד שב־nocturne/scripts/ באותו יום. אם סקריפט ישתנה, ההסבר כאן עלול לפגר אחריו. הקוד תמיד קובע.

למי המסמך ואיך לקרוא אותו

המסמך מניח שאינך קורא shell ואינך מכיר את llama.cpp מבפנים. לכל סקריפט יש אותו מבנה: קודם התמונה הגדולה (איפה הוא יושב בתהליך, על איזו מכונה מריצים אותו, מה הוא צריך כדי לעבוד ומה הוא משאיר אחריו), אחר כך איך מפעילים, אחר כך הזרימה בשפה פשוטה, ולבסוף שורה אחרי שורה: הקוד בעמודה אחת וההסבר לידו. שורות שהן יחידה אחת (למשל בלוק if של ארבע שורות) מוסברות יחד.

לפני הסקריפטים יש שני פרקים קצרים: מילון shell שמסביר פעם אחת את הסימנים שחוזרים בכל הסקריפטים, ומפת התהליך שמראה איך הסקריפטים משתלבים זה בזה. אם בהסבר של שורה מופיע ביטוי לא מוכר, הוא כנראה במילון.

מילון shell: הסימנים שחוזרים בכל סקריפט

סקריפט shell הוא קובץ טקסט שבו כל שורה היא פקודה שהיית יכול להקליד בטרמינל. המחשב קורא אותו מלמעלה למטה ומבצע שורה אחר שורה. הסקריפטים שלנו כתובים ל־bash, ונשמרת תאימות ל־bash 3.2 הישן של macOS (ולכן אין בהם כמה תכונות מודרניות). הטבלה מסבירה את הסימנים שתפגוש שוב ושוב. אפשר לדלג עליה ולחזור אליה כשצריך.

מה כתובמה זה אומר
#!/usr/bin/env bashהשורה הראשונה בכל סקריפט. אומרת למערכת ההפעלה "הרץ את הקובץ הזה עם bash". היא לא פקודה אלא הוראה למי שמפעיל את הקובץ.
# textהערה. כל מה שאחרי # עד סוף השורה מתעלמים ממנו. ההערות בראש כל סקריפט הן התיעוד שלו, ובכל הסקריפטים שלנו הן מכילות דוגמאות הפעלה.
set -euo pipefailשלוש הגנות שמופעלות בתחילת הריצה. -e: אם פקודה נכשלת, עצור מיד במקום להמשיך בשקט. -u: שימוש במשתנה שלא הוגדר הוא שגיאה (ולכן תראה הרבה ${X:-}, שמעניק ברירת מחדל ריקה). pipefail: בשרשרת של פקודות המחוברות ב־|, כישלון של אחת מהן נחשב כישלון של כולן.
NAME="value"הגדרת משתנה. בלי רווחים סביב =. מכאן והלאה $NAME או ${NAME} מתחלפים בערך.
"${NAME:-default}"הערך של NAME, ואם הוא לא הוגדר או ריק, השתמש ב־default. זה המנגנון שבו כל הסקריפטים נותנים ערכי ברירת מחדל שאפשר לדרוס מבחוץ: PP=512 scripts/08_r0.sh ... קובע PP לריצה הזו בלבד.
"${NAME:+text}"ההפך: אם NAME לא ריק, תן text; אחרת כלום. משמש להוסיף פסיק או ארגומנט רק כשיש מה להוסיף.
$1 $2 "${1:-}"הארגומנטים שניתנו לסקריפט בשורת הפקודה, לפי סדרם. scripts/02_run_local.sh MTL0 נותן $1=MTL0. $0 הוא שם הסקריפט עצמו. בתוך פונקציה, $1 הוא הארגומנט של הפונקציה.
"$@" "$*"כל הארגומנטים ביחד. "$@" שומר על כל ארגומנט כיחידה נפרדת (נכון להעברה הלאה לפקודה אחרת); "$*" מדביק אותם למחרוזת אחת (נכון להדפסה).
$(command)הרץ את הפקודה ושים את מה שהיא הדפיסה במקום הביטוי. HOST="$(hostname -s)" שומר את שם המחשב במשתנה.
$(( 3 + 4 ))חשבון שלמים. $((T1-T0)) הוא הפרש שניות.
. file (source)"טען את הקובץ הזה לתוך הריצה הנוכחית", כאילו שורותיו נכתבו כאן. כך כל סקריפט מקבל את המשתנים והפונקציות של common.sh.
if [ test ]; then ... else ... fiתנאי. הסוגריים המרובעים הם פקודת בדיקה. בדיקות נפוצות: -f X קובץ קיים, -d X תיקייה קיימת, -x X קובץ בר הרצה, -s X קובץ לא ריק, -n "$X" מחרוזת לא ריקה, -z "$X" מחרוזת ריקה, "$A" = "$B" שוויון מחרוזות, "$A" -eq 0 שוויון מספרי, ! שלילה.
A && Bהרץ B רק אם A הצליחה. נפוץ כקיצור ל־if: [ -n "$DEVICE" ] && ARGS+=(--device "$DEVICE") פירושו "אם יש device, הוסף אותו לארגומנטים".
A || Bהרץ B רק אם A נכשלה. command -v cmake || die "..." פירושו "אם אין cmake, עצור עם הודעה". || true פירושו "גם אם נכשל, אל תעצור" (עוקף את set -e במקום אחד).
case "$X" in a) ... ;; b|c) ... ;; *) ... ;; esacבחירה לפי ערך. אם X הוא a בצע את הענף הראשון; b או c את השני; * הוא "כל השאר". תבניות כמו *:* פירושן "מכיל נקודתיים". כל ענף מסתיים ב־;;.
name() { ...; }הגדרת פונקציה. לא מבצעת כלום עד שקוראים לה בשמה. local X בתוך פונקציה מגדיר משתנה שחי רק בתוכה.
ARGS=(a b c) ARGS+=(d) "${ARGS[@]}"מערך. הסקריפטים בונים את שורת הפקודה של llama.cpp חתיכה אחר חתיכה במערך ARGS, ובסוף מריצים "$BIN/llama-cli" "${ARGS[@]}". היתרון: ארגומנט עם רווח (כמו prompt) נשאר יחידה אחת.
cmd > file cmd >> fileהפניית הפלט הרגיל (stdout) לקובץ. > דורס, >> מוסיף בסוף.
cmd 2> file cmd 2>&1 cmd >/dev/null 2>&1לכל תוכנית יש שני ערוצי פלט: רגיל (1) ושגיאות (2). 2> מפנה את ערוץ השגיאות. 2>&1 מאחד את השגיאות לתוך הפלט הרגיל. /dev/null הוא "פח": הפלט נזרק. חשוב אצלנו: llama.cpp כותב את כל לוג הטעינה (איזו שכבה לאיזה מכשיר) לערוץ השגיאות, ולכן הסקריפטים שומרים אותו בנפרד כ־stderr.log.
cmd1 | cmd2צינור: הפלט של הראשונה הופך לקלט של השנייה. grep X | head -1 = "השורות שמכילות X, ומהן רק הראשונה".
cmd | tee filetee מעביר את מה שהוא מקבל גם למסך וגם לקובץ. כך רואים את הריצה בזמן אמת וגם שומרים לוג.
${PIPESTATUS[0]}בצינור, קוד היציאה שרואים הוא של הפקודה האחרונה (tee, שתמיד מצליחה). PIPESTATUS[0] נותן את קוד היציאה של הפקודה הראשונה, כלומר של llama.cpp עצמו. 0 = הצלחה, כל מספר אחר = כישלון.
$?קוד היציאה של הפקודה האחרונה שרצה.
exit 1סיים את הסקריפט עם קוד כישלון. exit 0 = הצלחה. מי שמריץ סקריפט בתוך סקריפט בודק את הקוד הזה.
exec cmd"החלף את הסקריפט בפקודה הזו". מהרגע הזה אין יותר סקריפט, יש רק את התוכנית. משמש כשהסקריפט הוא רק עטיפה שמכינה שרת ואז נעלמת (03, 13).
command -v tool"האם הכלי הזה מותקן ואיפה". מצליח אם קיים. בשילוב >/dev/null 2>&1 זו בדיקה שקטה.
cmd <<EOF ... EOFheredoc: כל הטקסט עד השורה EOF נשלח כקלט לפקודה (או נכתב לקובץ אם יש > file). עם <<'EOF' (גרשיים) הטקסט נלקח כפי שהוא, בלי להחליף $X בערכים; בלי גרשיים, המשתנים מוחלפים. כך הסקריפטים כותבים קובצי JSON ומריצים קטעי python מתוך bash.
printf 'fmt' args echo textהדפסה. printf עם תבנית (%s = שים מחרוזת כאן, \n = שורה חדשה), echo פשוט מדפיס. printf '%q ' מדפיס כל ארגומנט בצורה שאפשר להדביק חזרה לטרמינל (עם escaping), וכך נשמרת הפקודה המדויקת ב־cmd.txt.
grep -E 'pattern' fileמצא שורות שמתאימות לתבנית (ביטוי רגולרי). דגלים: -q שקט, רק הצלחה/כישלון; -o הדפס רק את החלק שהתאים; -c ספור; -i התעלם מאותיות גדולות/קטנות; -m1 עצור אחרי ההתאמה הראשונה; -F חפש טקסט מילולי ולא תבנית. בתבנית: | "או", [0-9]+ "ספרה אחת או יותר", .* "כל דבר", ^ תחילת שורה, $ סופה.
sed -E 's/from/to/'החלפת טקסט בכל שורה. s#a#b# זהה, רק עם מפריד אחר כדי שלא יתנגש עם / בתבנית. \1 בחלק ה־"to" הוא "הקטע שנתפס בסוגריים בתבנית". sed 's/^/ /' מוסיף הזחה בתחילת כל שורה (להדפסה נאה).
awk '{print $1}'עיבוד לפי עמודות. $1 העמודה הראשונה, $NF האחרונה, $(NF-1) הלפני אחרונה. awk '/pattern/{...}' מבצע רק בשורות שמתאימות.
cut -d' ' -f1 tr 'a' 'b' head -1 / tail -1 / tail -n +2cut חותך עמודה לפי מפריד. tr מחליף תו בתו (או מוחק עם -d). head -1 השורה הראשונה, tail -1 האחרונה, tail -n +2 הכול חוץ מהראשונה (דילוג על כותרת).
basename path dirname pathשם הקובץ בלבד / התיקייה בלבד. basename "$MODEL" נותן את שם קובץ המודל בלי הנתיב.
mkdir -p dirצור תיקייה, כולל תיקיות אב, ואל תתלונן אם כבר קיימת.
date -u +%Y%m%d-%H%M%Sהתאריך והשעה ב־UTC בפורמט נתון, למשל 20261024-091500. זה הבסיס לכל שמות ה־run.
${X//,//} ${X%%,*} ${X#*,}עיבוד מחרוזת מובנה. ${X//a/b} מחליף כל a ב־b (אצלנו: פסיק ל־slash). ${X%%,*} מוחק מהפסיק הראשון עד הסוף (נשאר החלק שלפני הפסיק). ${X#*,} מוחק עד הפסיק הראשון כולל (נשאר החלק שאחרי). כך 1,0 מתפרק ל־1 ו־0.
IFS=','התו שלפיו bash מפרק מחרוזת למילים. ברירת המחדל היא רווח. שינוי ל־, מאפשר לולאה על רשימה מופרדת בפסיקים, ואחר כך משחזרים את הערך הקודם.
cmd & PID=$! wait $PIDהרצה ברקע. & לא מחכה לסיום; $! הוא מספר התהליך שהורץ; wait מחכה לו. kill/pkill הורגים תהליך לפי מספר/שם; pgrep בודק אם קיים.
env VAR=x cmd export VAR unset VARמשתני סביבה: ערכים שתוכנית־בת רואה. env VAR=x cmd מריץ את cmd עם המשתנה; export הופך משתנה רגיל לכזה שתוכניות־בת רואות; unset מוחק אותו.
script -q -a -F file cmdכלי מערכת שמריץ תוכנית בתוך "טרמינל מדומה" ומקליט הכול לקובץ. אנחנו משתמשים בו כי ggml-rpc-server ו־llama-server לא מדפיסים כלום כשהפלט שלהם הולך לצינור (buffering); עם טרמינל מדומה השורות מופיעות מיד, גם על המסך וגם בלוג.
shellcheck disable=...הערה לכלי בדיקת הקוד shellcheck, שיתעלם מאזהרה מסוימת. אין לזה השפעה על הריצה.

מפת התהליך: מי מריץ מה, איפה ומתי

התוכניות שהסקריפטים מפעילים

הסקריפטים עצמם לא מחשבים כלום. הם מכינים ארגומנטים, מפעילים תוכנית חיצונית, ושומרים את הפלט. אלה התוכניות:

תוכניתמאיפהמה היא עושהמי מפעיל אותה
llama-clillama.cppטוענת מודל GGUF, מקבלת prompt, מייצרת טקסט. עם --rpc היא יודעת לשלוח חלק מהשכבות למכונה אחרת. עם --verbose היא מדפיסה איזו שכבה הלכה לאיזה מכשיר.02, 04, 08 (placement), 10

התמונה הגדולה

התמונה הגדולה

זה הסקריפט של הטענה המרכזית בעבודה (T2, O2, O4.1): "מודל שלא נכנס במכונה אחת רץ על שתיים". כדי שהטענה תהיה מדויקת צריך להגדיר מה זה "נכנס", ולכן יש שתי מדיניות זיכרון מוצהרות. P1: כל השכבות על מאיצים, בלי שמנוע ההסקה יתאים את עצמו (-ngl 99 -fit off); נכנס רק אם כל השכבות הועברו. P2: רץ בכלל, גם אם llama.cpp השאיר שכבות ב־CPU (-fit on); רושמים כמה שכבות הגיעו ל־GPU. כל ניסיון, מקומי או מפוצל, נרשם באותו פורמט עם פסק דין אוטומטי. הסקריפט גם כולל את הדרך לחזור על שאלת ה־TP של חן (SPLIT_MODE=tensor) עם לוג.

מתילכל מודל שרוצים לטעון: קודם local על כל מכונה (להוכיח שלא נכנס), ואז split מול worker (להוכיח שנכנס).
דורשllama-cli; נתיב למודל כארגומנט (לא מ־.env, כי בודקים כמה מודלים); במצב split גם worker פעיל.

הגנהwatchdog שהורג את הריצה אחרי TIMEOUT_S (ברירת מחדל 900 שניות). נכתב אחרי שריצה שנכנסה ל־swap המשיכה 17 דקות למרות ניסיון לעצור אותה.

11_load.py: עומס מקבילי כל מכונה (הלקוח)

התמונה הגדולה

O4.2 שואל "איפה כל תצורה מפסיקה לגדול". כדי לענות צריך לשלוח הרבה בקשות במקביל לשרת ולמדוד כמה טוקנים לשנייה יוצאים בסך הכול, לא לבקשה בודדת. הסקריפט הזה הוא מחולל העומס. הוא מדבר עם llama-server (שמרימים ב־13) דרך HTTP. הרעיון החשוב: אותו כלי מודד גם שרת מפוצל אחד (requester + worker, כתובת אחת) וגם שני שרתים עצמאיים (אחד בכל מכונה, שתי כתובות, הבקשות מתחלקות ביניהם). כך משווים "פיצול" מול "replicas" בתנאים זהים, וזו היפותזה H2.

למה פייתון stdlib בלבדכדי לרוץ על /usr/bin/python3 של macOS (גרסה 3.9) בלי להתקין כלום. אין requests, אין asyncio; threads ו־urllib.
